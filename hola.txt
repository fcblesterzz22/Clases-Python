1. #TRAER DATOS DEL REPOSITORIO
git pull

2. VER LA RAMA
git branch

3. PASAR DE UNA RAMA A OTRA RAMA (osea rama 1 a rama 2 o bien de rama 2 a rama 1)
git checkout nombre_de_la_rama

4. GUARDAR CAMBIOS AL REPOSITORIO
git add .

5. VER ESTADO DE LA RAMA
git status

6. COLOCAR UN COMENTARIO EN EL CAMBIO REALIZADO
git commit -m "aqui adentro va el comentario" 

7. SUBIR CAMBIOS AL REPOSITORIO
git push
